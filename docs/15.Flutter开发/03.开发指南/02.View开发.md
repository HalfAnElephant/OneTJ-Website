## View 概述

在 OneTJ 的分层 MVVM 架构中，View 负责渲染 UI 并响应用户交互：通过 `AnimatedBuilder` / `ListenableBuilder` 监听 ViewModel 状态，通过 `StreamSubscription` 订阅 `UiEvent` 处理一次性事件。

## View 结构

### 基本结构

```dart
class DashboardView extends StatefulWidget {
  const DashboardView({super.key});

  @override
  State<DashboardView> createState() => _DashboardViewState();
}

class _DashboardViewState extends State<DashboardView> {
  late final DashboardViewModel _viewModel;
  StreamSubscription<UiEvent>? _eventSub;

  @override
  void initState() {
    super.initState();
    _viewModel = DashboardViewModel();
    _eventSub = _viewModel.events.listen(_handleEvent);
    _viewModel.load();
  }

  void _handleEvent(UiEvent event) {
    if (!mounted) return;
    if (event is ShowSnackBarEvent) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(event.message ?? '')),
      );
    }
  }

  @override
  void dispose() {
    _eventSub?.cancel();
    _viewModel.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Dashboard')),
      body: AnimatedBuilder(
        animation: _viewModel,
        builder: (context, _) => _buildBody(context),
      ),
    );
  }
}
```

## 状态监听

### AnimatedBuilder / ListenableBuilder

ViewModel 是 `ChangeNotifier`，使用 `AnimatedBuilder` 或 `ListenableBuilder` 监听状态变化：

```dart
body: AnimatedBuilder(
  animation: _viewModel,
  builder: (context, _) {
    return _viewModel.loading
        ? const LinearProgressIndicator()
        : _buildContent(_viewModel.data);
  },
)
```

需要同时监听多个 `Listenable`（如 ViewModel + ThemeChangeNotifier）时：

```dart
body: AnimatedBuilder(
  animation: Listenable.merge([_viewModel, _themeChangeNotifier]),
  builder: (context, _) => _buildBody(context),
)
```

### 事件订阅

一次性事件通过 `events`（`Stream<UiEvent>`）处理：

```dart
_eventSub = _viewModel.events.listen((event) {
  if (!mounted) return;

  if (event is ShowSnackBarEvent) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(event.message ?? '')),
    );
  } else if (event is AppUpdateAvailableEvent) {
    _showAppUpdateDialog(event.updateInfo);
  }
});
```

## 加载状态

### 三种状态：加载中、成功、空

```dart
Widget _buildBody(BuildContext context) {
  if (_viewModel.timetableLoading || _viewModel.calendarLoading) {
    return const LinearProgressIndicator();
  }
  final entries = _viewModel.buildUpcomingEntries();
  if (entries.isEmpty) {
    return const Center(child: Text('暂无课程'));
  }
  return _buildUpcomingSection(entries);
}
```

## UI 事件处理

### 按钮点击

```dart
FunctionGrid(
  onTimetableTap: () => context.go(RoutePaths.homeTimetable),
  onGradesTap: () => context.push(RoutePaths.homeGrades),
)
```

### 路由导航约定

- 主页 Shell 内的一级页面切换使用 `context.go`
- Shell 外详情页使用 `context.push`（保留返回主页的栈）

```dart
// Shell 内一级页面
onTimetableTap: () => context.go(RoutePaths.homeTimetable),

// Shell 外详情页
onGradesTap: () => context.push(RoutePaths.homeGrades),
```

## 最佳实践

### 1. 使用 const 构造函数

```dart
const Text('Hello')
const SizedBox(height: 16)
const Icon(Icons.home)
```

### 2. 提取 Widget

```dart
// ✅ 好
class StudentInfoCard extends StatelessWidget {
  final StudentInfoData info;
  const StudentInfoCard({super.key, required this.info});

  @override
  Widget build(BuildContext context) => Card(...);
}

// ❌ 不好
Widget buildStudentCard(StudentInfoData info) => Card(...);
```

### 3. 使用主题颜色

```dart
Container(
  color: Theme.of(context).colorScheme.primaryContainer,
  child: Text(
    'Hello',
    style: TextStyle(
      color: Theme.of(context).colorScheme.onPrimaryContainer,
    ),
  ),
)
```

### 4. 检查 mounted 状态

```dart
_eventSub = _viewModel.events.listen((event) {
  if (!mounted) return;
  // 处理事件
});
```

### 5. 及时取消订阅

```dart
@override
void dispose() {
  _eventSub?.cancel();
  _viewModel.dispose();
  super.dispose();
}
```

## 示例：完整的 View

```dart
class GradesView extends StatefulWidget {
  const GradesView({super.key});

  @override
  State<GradesView> createState() => _GradesViewState();
}

class _GradesViewState extends State<GradesView> {
  late final GradesViewModel _viewModel;
  StreamSubscription<UiEvent>? _eventSub;

  @override
  void initState() {
    super.initState();
    _viewModel = GradesViewModel();
    _eventSub = _viewModel.events.listen(_handleEvent);
    _viewModel.load();
  }

  void _handleEvent(UiEvent event) {
    if (!mounted) return;
    if (event is ShowSnackBarEvent) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text(event.message ?? '')),
      );
    }
  }

  @override
  void dispose() {
    _eventSub?.cancel();
    _viewModel.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('成绩')),
      body: AnimatedBuilder(
        animation: _viewModel,
        builder: (context, _) {
          if (_viewModel.loading) {
            return const Center(child: CircularProgressIndicator());
          }
          return _buildContent(_viewModel.data);
        },
      ),
    );
  }

  Widget _buildContent(GradesViewData? data) {
    if (data == null) return const Center(child: Text('暂无数据'));
    return ListView.builder(
      itemCount: data.items.length,
      itemBuilder: (context, index) => ListTile(
        title: Text(data.items[index].title),
      ),
    );
  }
}
```
